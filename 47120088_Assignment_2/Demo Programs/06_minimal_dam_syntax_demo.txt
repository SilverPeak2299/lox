// 06_minimal_dam_syntax_demo.txt
// Minimal example just to demonstrate dam syntax clearly.

[8.0, 0.0, 0.0, 0.0, 5.0, 0.0, 0.0, 3.0, 0.0, 0.0]

// Simple inflow river as a catchment
river_in = (5000);

// Dam inflow is this river (no extra catchment here, just the river).

// Very simple dam rule: more release when fill is high.
demo_dam = river_in + dam(0.5, 1000,
  if fill > 0.6 flow 0.7 else
  flow 0.3
);

// Expose the dam outflow as a named river.
river_out = demo_dam;
