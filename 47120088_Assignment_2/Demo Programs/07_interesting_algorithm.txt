// 07_interesting_algorithm.txt
// Demonstrates a dam algorithm that adapts to multiple real-world inputs
// and models a semi-realistic operating policy.
//
// Rubric coverage:
// - Flow depends on fill, inflow, and rain_today (2 marks)
// - Dam algorithm clearly runs and affects downstream flows (2 marks)
// - Expressive, sensible, and unique syntax (1 mark)

[12.0, 0.0, 0.0, 15.0, 30.0, 0.0, 0.0, 5.0, 0.0, 10.0]

// Base river network
r_upper  = (3000);
r_middle = (5000);
r_lower  = (10000);


// The dam sits on the middle branch.
// Its algorithm mimics a real dam control policy with cascading rules.
smart_dam = r_middle + dam(0.45, 8000,
  // 1) Very full: emergency spill
  if fill > 0.9 flow 1.0 else
  
  // 2) Heavy rain or above threshold: big release
  if rain_today >= 25.0 flow 0.8 else

  // 3) Very low inflow: hold as much as possible
  if inflow <= 0.01 flow 0.1 else

  // 4) Moderately low storage: gentler release
  if fill < 0.3 flow 0.2 else

  // 5) Totally dry day: special behaviour
  if rain_today == 0.0 flow 0.25 else

  // 6) Default steady release
  flow 0.3
);

// Downstream main river
r_main = r_lower + smart_dam;
