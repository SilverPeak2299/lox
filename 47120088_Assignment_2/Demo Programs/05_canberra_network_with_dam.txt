// 05_network_with_dam.txt
// Canberra-style network with a dam in the middle of the system.

[11.4, 0.0, 5.0, 0.0, 0.0, 12.0, 0.0, 0.0, 3.0, 9.0]

// Upstream rivers as catchments
upper_molo = (3000);
quean      = (1800);
jerra      = (1200);

// A simple upstream Googong river *before* the dam
googong_upstream = (4000) + upper_molo;

// Dam on the Googong branch with a small decision-list algorithm.
// Inline catchment definition
googong_dam = googong_upstream + (2000) + dam(0.3, 5000,
  if fill > 0.8 flow 0.8 else
  if inflow > 0.04 flow 0.5 else
  if rain_today > 8.0 flow 0.6 else
  flow 0.2
);

// Downstream Googong river is dam outflow.
googong = googong_dam;

// Central Molonglo: all branches combining.
central_molo = upper_molo + googong + quean + jerra;

// Lower Molonglo downstream.
lower_molo = central_molo;
