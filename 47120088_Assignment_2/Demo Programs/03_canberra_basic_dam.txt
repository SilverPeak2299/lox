// 03_basic_dam.txt
// Single simple dam on the Googong branch.
// Algorithm depends only on dam level (fill).

[11.4, 4000.0, 50.0, 0.0, 0.0, 2.0, 0.2, 0.2, 0.2, 0.0]

// Upstream rivers as catchments
upper_molo = (3000);
quean      = (1800);
jerra      = (1200);

// All water going into the Googong dam.
// Shows chaining a river AND an inline catchment into a dam inflow.
googong_dam_inflow = upper_molo + (6000);

// Dam on the Googong branch.
// Dam output = fraction of fill
googong_dam = dam(0.8, 10000,
  if fill > 0.8 flow 0.7 else
  flow 0.3
);

// River "googong" is ONLY the water coming out of the dam.
googong = googong_dam;

// Central and lower rivers use the dammed Googong.
central_molo = googong + quean + jerra;
lower_molo   = central_molo;
