// 04_canberra_dam_multi_factor.txt
// Smart dam on Googong. Its behaviour depends on:
//   - fill (dam level)
//   - inflow
//   - rain_today

[11.4, 0.0, 0.4, 0.0, 0.0, 20.0, 0.2, 30, 0.2, 0.0]

// Upstream rivers
upper_molo = (3000);
googong_in = (6000);
quean      = (1800);
jerra      = (1200);

// Smart dam:
//  - If very full -> release a lot.
//  - Else if inflow is big -> moderate release.
//  - Else if today is a big rain event -> extra release.
//  - Else -> small trickle.
googong_dam = dam(0.4, 8000,
  if fill > 0.85 flow 0.9 else
  if inflow > 0.05 flow 0.5 else
  if rain_today > 15 flow 0.7 else
  flow 0.2
);

googong = googong_dam;

// Downstream rivers
central_molo = upper_molo + googong + quean + jerra;
lower_molo   = central_molo;
