// 04_dam_multi_factor.txt
// Smart dam on the Googong branch.
// Behaviour depends on: fill, inflow, and rain_today.

[10.0, 0.0, 3.0, 0.0, 0.0, 18.0, 0.0, 0.0, 7.0, 22.0]

// Upstream rivers as catchments
upper_molo = (3000);
quean      = (1800);
jerra      = (1200);

// Dam inflow: upper_molo plus an inline Googong catchment.
googong_dam_inflow = upper_molo + (6000);

// Smart dam:
//  - If very full -> strong release.
//  - Else if inflow high -> moderate release.
//  - Else if today is a big rain day -> extra release.
//  - Else -> small base release.
googong_dam =  googong_dam_inflow + dam(0.4, 8000,
  if fill > 0.85 flow 0.9 else
  if inflow > 0.05 flow 0.5 else
  if rain_today > 15.0 flow 0.7 else
  flow 0.2
);

googong = googong_dam;

// Downstream rivers
central_molo = googong + quean + jerra;
lower_molo   = central_molo;
